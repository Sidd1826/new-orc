"""
mock_gateway.py — a fake HSBC MHA Gateway for LOCAL testing only.

Speaks exactly what gateway_client.py sends and processor.py reads:
  GET  /api/v1/accounts/demand-deposit   body {"sensitiveData":[{"key","value"}]}
       -> data.accounts[0].availableBalance.amount       (config.BALANCE_PATH)
  GET  /api/v1/transactions/history      ?transactionStartDate=YYYY-MM-DD + same body
       -> data.accountInformation.transactionDetails[]  (RRN in a narrative)
  POST /api/v1/accounts/apply-hold       {"applyHoldDetails": {...}} -> data.status
  POST /api/v1/upi/secure                {"systemId", "payload": {"rrn", "accNo", ...}}
       -> data.{pyAccNo, pyIfscCode, bankName, pyFname, txnDate}; {} if unknown
  POST /api/v1/search                    {"LPUTRN": <utr>, "limit"} -> rows[]
  POST /api/v1/mha-response              raw base64 body -> data.raw = encrypted ack

Behaviour is derived from the 12-digit ACCOUNT so one mock covers every path:
  FIXTURE_ACCOUNT  known account: a UPI txn (RRN 630818826981) and an IMPS
                   txn (RRN 630815826944) on 2026-11-04; any other RRN -> 02
  ABSENT_ACCOUNT   HTTP 404 (account absence -> 01)
  EMPTY_ACCOUNT    HTTP 200, empty transaction list (no records -> 01)
  FAILING_ACCOUNT  HTTP 503 (gateway failure: retried, never a business code)
  MALFORMED_ACCOUNT HTTP 200 with an error envelope (retried, never 01)
These are test fixtures, never production routing.

Run:  uvicorn mock_gateway:app --host 0.0.0.0 --port 8000
"""
from __future__ import annotations

import uuid
from typing import Any, Optional

from fastapi import Body, FastAPI, Request
from fastapi.responses import JSONResponse

app = FastAPI(title="MOCK HSBC MHA Gateway")

FIXTURE_ACCOUNT   = "003240868006"
ABSENT_ACCOUNT    = "999999999999"
EMPTY_ACCOUNT     = "111111111111"
FAILING_ACCOUNT   = "500500500500"
MALFORMED_ACCOUNT = "600600600600"

UPI_RRN  = "630818826981"
IMPS_RRN = "630815826944"


def _txn(narratives: list[str], amount: str, ts: str) -> dict:
    return {"postStatusDescription": "APPROVED", "systemCode": "DD",
            "transactionAmount": {"amount": amount, "currency": "INR"},
            "transactionId": {"executionTimestamp": ts},
            "transactionDescription": [{"transactionNarrative": n} for n in narratives]}


HISTORY: dict[str, list[dict]] = {
    FIXTURE_ACCOUNT: [
        _txn(["UPI", UPI_RRN, "P2A"], "-50000.00", "2026-11-04T18:44:00+05:30"),
        _txn(["IMPS", IMPS_RRN, "HDFC/500021522001"], "-900.00", "2026-11-04T15:11:00+05:30"),
        _txn(["UPI", "630815826900"], "-120.00", "2026-11-04T09:00:00+05:30"),
    ],
    EMPTY_ACCOUNT: [],
}
BALANCES: dict[str, str] = {FIXTURE_ACCOUNT: "76000.00", EMPTY_ACCOUNT: "0.00"}
UPI_BENE: dict[str, dict] = {
    UPI_RRN: {"pyAccNo": "50100200300", "pyIfscCode": "HDFC0000001", "bankName": "HDFC Bank",
              "pyFname": "MOCK BENEFICIARY", "pyFvaddr": "mock@hdfc",
              "txnDate": "2026/11/04 18:44:00"},
}
CBDOC_ROWS: dict[str, list[dict]] = {}          # LPUTRN -> rows (NEFT/RTGS legs)


def _account(body: Any) -> Optional[str]:
    try:
        return body["sensitiveData"][0]["value"]
    except Exception:
        return None


def _envelope(data: Any) -> dict:
    return {"requestId": str(uuid.uuid4()), "data": data}


def _special(account: Optional[str]) -> Optional[JSONResponse]:
    if account is None:
        return JSONResponse({"detail": "sensitiveData missing"}, status_code=400)
    if account == ABSENT_ACCOUNT:
        return JSONResponse({"errors": [{"code": "ACCOUNT_NOT_FOUND"}]}, status_code=404)
    if account == FAILING_ACCOUNT:
        return JSONResponse({"detail": "upstream unavailable"}, status_code=503)
    if account == MALFORMED_ACCOUNT:
        return JSONResponse({"errors": [{"code": "E500", "message": "internal error"}]})
    return None


@app.get("/health")
async def health():
    return {"status": "ok", "service": "MOCK-gateway"}


@app.get("/api/v1/accounts/demand-deposit")
async def demand_deposit(hubCustomerNumber: str = "", body: Any = Body(default=None)):
    account = _account(body)
    if (err := _special(account)) is not None:
        return err
    bal = BALANCES.get(account, "100000.00")
    return _envelope({"accounts": [{"accountStatusDesc": "ACTIVE",
                                    "availableBalance": {"currencyCode": "INR", "amount": bal},
                                    "currentBalance": {"currencyCode": "INR", "amount": bal}}]})


@app.get("/api/v1/transactions/history")
async def transaction_history(accountNumber: str = "", transactionStartDate: str = "",
                              body: Any = Body(default=None)):
    account = _account(body)
    if (err := _special(account)) is not None:
        return err
    txns = [t for t in HISTORY.get(account, HISTORY[FIXTURE_ACCOUNT])
            if t["transactionId"]["executionTimestamp"][:10] >= transactionStartDate]
    return _envelope({"accountInformation": {"accountCurrencyCode": "INR", "moreTxns": "N",
                                             "transactionDetails": txns}})


@app.post("/api/v1/accounts/apply-hold")
async def apply_hold(body: dict = Body(...)):
    d = body.get("applyHoldDetails", {})
    return _envelope({"status": "SUCCESS", "holdReference": f"HOLD-{uuid.uuid4().hex[:10]}",
                      "heldAmount": d.get("holdAmount"), "accountNumber": d.get("accountNumber")})


@app.post("/api/v1/upi/secure")
async def upi_secure(body: dict = Body(...)):
    rrn = str((body.get("payload") or {}).get("rrn", ""))
    return _envelope(UPI_BENE.get(rrn, {}))      # {} = the UPI switch has no beneficiary


@app.post("/api/v1/search")
async def search(body: dict = Body(...)):
    rows = CBDOC_ROWS.get(str(body.get("LPUTRN", "")), [])
    return {"requestId": str(uuid.uuid4()), "filters_used": body,
            "row_count": len(rows), "rows": rows}


@app.post("/api/v1/mha-response")
async def mha_response(request: Request):
    import encryption                             # needs MHA_RESPONSE_KEY_B64
    payload = encryption.decrypt_payload((await request.body()).decode("utf-8"))
    ack = {"message": f"Response has been successfully added. {payload.get('job_id')}",
           "statuscode": "00"}
    return _envelope({"raw": encryption.encrypt_payload(ack)})
