-- ---- I4C-MHA sheet fixtures (3160823000011-16 + corrected 0015) -------------
-- Transaction date 2026-11-04, layer 0, Debit, HSBC bank code 10, disputed =
-- amount. Expected with mock_gateway's fixture accounts:
--   3160823000011   003240868006 RRN 630818826981 -> 00 Money Transfer To (UPI record)
--   3160823000012   999999999999 (mock: HTTP 404) -> 01 Record not found
--   3160823000013   3240868006   (10 digits)      -> 12 Invalid account number, no gateway call
--   3160823000014   3240A68006   (alphanumeric)   -> 12 Invalid account number, no gateway call
--   3160823000015   999999999999 RRN 999999999999 -> 01: the account check wins; never 02
--   3160823000015C  003240868006 RRN 999999999999 -> 02 Invalid RRN (corrected fixture:
--                   a known account whose history lacks that RRN)
--   3160823000016   003240868006 RRN 630815826944 -> 31 Pending (the bank record is IMPS;
--                   transaction_type 'IMPS' is kept as metadata only)
WITH v(ack, acct, rrn, amt, tm, ttype) AS (VALUES
    ('3160823000011',  '003240868006', '630818826981', 50000.00, '18:44', 'Unknown'),
    ('3160823000012',  '999999999999', '630815826975',  1000.00, '15:52', 'Unknown'),
    ('3160823000013',  '3240868006',   '630815826952',   900.00, '15:16', 'Unknown'),
    ('3160823000014',  '3240A68006',   '630815826952',   900.00, '15:16', 'Unknown'),
    ('3160823000015',  '999999999999', '999999999999',  1000.00, '15:12', 'Unknown'),
    ('3160823000015C', '003240868006', '999999999999',  1000.00, '15:12', 'Unknown'),
    ('3160823000016',  '003240868006', '630815826944',   900.00, '15:11', 'IMPS')),
c AS (INSERT INTO complaints (acknowledgement_no, sub_category, status)
      SELECT ack, 'UPI Related Frauds', 'OPEN' FROM v
      ON CONFLICT (acknowledgement_no) DO NOTHING RETURNING id, acknowledgement_no),
i AS (INSERT INTO instruments (complaint_id, requestor, payer_bank, payer_bank_code,
          mode_of_payment, transaction_type, wallet, payer_mobile_number,
          payer_account_number, state, district)
      SELECT c.id, 'I4C-MHA', 'HSBC', '10', 'Debit', v.ttype, 'Phonepe', '919798868768',
             v.acct, 'DELHI', 'South Delhi'
      FROM c JOIN v ON v.ack = c.acknowledgement_no RETURNING id, complaint_id)
INSERT INTO incidents (instrument_id, merchant_id, amount, disputed_amount, layer,
                       transaction_date, transaction_time)
SELECT i.id, v.rrn, v.amt, v.amt, 0, DATE '2026-11-04', v.tm::time
FROM i JOIN c ON c.id = i.complaint_id JOIN v ON v.ack = c.acknowledgement_no;
