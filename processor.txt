r"""
processor.py — the orchestration engine.

Trigger model: NO queue. Cloud Scheduler hits /process-batch every 1-2 min;
this module claims a batch of un-finished complaints with
`FOR UPDATE SKIP LOCKED`, leases them, and runs a resumable state machine.

State machine (persisted on complaint_processing.processing_state — a SIDECAR
table this service owns; the shared `complaints` table is never written):

  RECEIVED -> CLASSIFIED -> ENRICHED -> BALANCE_CHECKED
           -> HOLD_APPLIED -> RECONCILING -> RESOLVED
                                          \-> NEEDS_REVIEW (terminal, human)

Money rule (from your prose):
  D = total disputed amount, B = available balance.
  B >= D  -> hold D, done.
  B <  D  -> hold B, remaining R = D - B, then trace R through CBDOC/BigQuery,
             collecting every debited transaction until reconciled (or layers
             exhausted -> NEEDS_REVIEW with whatever was collected).

Idempotency: apply-hold is guarded by a UNIQUE idempotency_key in
processing_steps, so a retry never double-holds a customer's funds.
"""

def gateway() -> GatewayClient:
    global _gateway
    if _gateway is None:
        _gateway = GatewayClient()
    return _gateway


def _now() -> datetime:
    return datetime.now(timezone.utc)


def _dig(d: Any, path: str) -> Any:
    """Read a dotted path out of a nested dict/list; None if missing.
    Numeric path segments index into lists, e.g. 'accounts.0.currentBalance.amount'."""
    cur = d
    for part in path.split("."):
        if isinstance(cur, list):
            try:
                cur = cur[int(part)]
            except (ValueError, IndexError):
                return None
        elif isinstance(cur, dict) and part in cur:
            cur = cur[part]
        else:
            return None
    return cur


def _usable_number(x: Any) -> bool:
    """True if x is a number or a non-empty numeric string."""
    if isinstance(x, bool):
        return False
    if isinstance(x, (int, float)):
        return True
    if isinstance(x, str):
        try:
            Decimal(x.strip())
            return x.strip() != ""
        except Exception:
            return False
    return False


def _deep_find_balance(obj: Any, container_key: str, amount_key: str = "amount",
                       _path: str = "") -> tuple[Any, str]:
    """Find the balance anywhere in a nested dict/list. Accepts `container_key`
    as EITHER a scalar (e.g. data.currentBalance = "76.21") OR an object
    (e.g. ...currentBalance.amount = 76.21). Returns (value, dotted_path)."""
    if isinstance(obj, dict):
        if container_key in obj:
            v = obj[container_key]
            if isinstance(v, dict) and amount_key in v and _usable_number(v[amount_key]):
                return v[amount_key], f"{_path}.{container_key}.{amount_key}".lstrip(".")
            if _usable_number(v):
                return v, f"{_path}.{container_key}".lstrip(".")
        for k, val in obj.items():
            r, fp = _deep_find_balance(val, container_key, amount_key,
                                       f"{_path}.{k}".lstrip("."))
            if r is not None:
                return r, fp
    elif isinstance(obj, list):
        for i, val in enumerate(obj):
            r, fp = _deep_find_balance(val, container_key, amount_key,
                                       f"{_path}.{i}".lstrip("."))
            if r is not None:
                return r, fp
    return None, ""


def _coerce_balance(x: Any) -> Optional[Decimal]:
    if x is None:
        return None
    try:
        return Decimal(str(x).strip())
    except Exception:
        return None


def _shape(obj: Any, depth: int = 3) -> Any:
    """Keys-only structural description (NO values) — safe to log/show."""
    if isinstance(obj, dict):
        if depth <= 0:
            return {k: "..." for k in obj}
        return {k: _shape(v, depth - 1) for k, v in obj.items()}
    if isinstance(obj, list):
        if not obj:
            return []
        head = _shape(obj[0], depth - 1)
        return [head, f"...(+{len(obj) - 1} more)"] if len(obj) > 1 else [head]
    return type(obj).__name__


# ===========================================================================
# Batch claim (the polling tick)
# ===========================================================================

def _bootstrap_processing_rows(s: Session) -> int:
    """Create a complaint_processing row for any complaint that doesn't have one
    yet (i.e. rows the OTHER service just inserted). Does NOT touch complaints.
    INSERT ... SELECT ... ON CONFLICT DO NOTHING is race-safe across instances.
    """
    res = s.execute(text(
        """
        INSERT INTO complaint_processing (complaint_id, acknowledgement_no, processing_state)
        SELECT c.id, c.acknowledgement_no, 'RECEIVED'
        FROM complaints c
        LEFT JOIN complaint_processing p ON p.complaint_id = c.id
        WHERE p.id IS NULL
        ON CONFLICT (complaint_id) DO NOTHING
        """
    ))
    s.commit()
    return res.rowcount or 0


def claim_batch(limit: int) -> list[int]:
    """Bootstrap missing sidecar rows, then claim up to `limit` non-terminal,
    lease-free processing rows. Returns the claimed complaint_processing ids.
    Nothing in the shared `complaints` table is written."""
    now = _now()
    with session_scope() as s:
        _bootstrap_processing_rows(s)

        rows = s.execute(
            select(ComplaintProcessing)
            .where(
                ComplaintProcessing.processing_state.notin_(tuple(TERMINAL_STATES)),
                ComplaintProcessing.attempt_count < config.MAX_ATTEMPTS,
                or_(ComplaintProcessing.lease_expires_at.is_(None),
                    ComplaintProcessing.lease_expires_at < now),
                or_(ComplaintProcessing.next_attempt_at.is_(None),
                    ComplaintProcessing.next_attempt_at <= now),
            )
            .order_by(ComplaintProcessing.created_at)
            .limit(limit)
            .with_for_update(skip_locked=True)
        ).scalars().all()

        lease = now + timedelta(seconds=config.LEASE_SECONDS)
        ids = []
        for p in rows:
            p.lease_expires_at = lease
            if not p.job_id:
                p.job_id = f"BANKS-{uuid.uuid4()}"
            ids.append(p.id)
        s.commit()
        if ids:
            log.info("claimed %d complaints", len(ids))
        return ids


def process_batch() -> dict[str, Any]:
    """One polling tick: claim a batch and process each in its own transaction."""
    ids = claim_batch(config.BATCH_SIZE)
    processed = []
    for pid in ids:
        state = _process_one(pid)
        processed.append({"processing_id": pid, "state": state})
    return {"claimed": len(ids), "processed": processed}


class AccountNotFound(Exception):
    """The gateway answered HTTP 404 for the payer account."""


class CbdocConflict(Exception):
    """CBDOC returned several different rows for one UTR; never pick one arbitrarily."""


def _process_one(processing_id: int) -> str:
    """Run the state machine for one complaint_processing row. Failures are
    recorded and the row is left for the next tick (until MAX_ATTEMPTS)."""
    try:
        with session_scope() as s:
            p = s.get(ComplaintProcessing, processing_id)
            if p is None or p.processing_state in TERMINAL_STATES:
                return p.processing_state if p else "MISSING"
            try:
                _run_state_machine(s, p)
            except AccountNotFound:
                s.rollback()
                incidents = p.complaint.instrument.incidents
                small = [i for i in incidents if _is_small(i)]
                if small:                   # the rollback may have dropped the stash
                    _stash_small_txns(s, p, small)
                _finalize_miscellaneous(s, p, [i for i in incidents if not _is_small(i)],
                                        "Record not found", MHA_RECORD_NOT_FOUND)
                _to_review(s, p, "account not found (gateway 404)")
            except CbdocConflict as e:
                s.rollback()
                _to_review(s, p, str(e))
            return p.processing_state
    except Exception as e:                      # noqa: BLE001 - record & defer
        log.exception("processing failed for processing_id=%s: %s", processing_id, e)
        _record_failure(processing_id, str(e))
        with session_scope() as s:
            p = s.get(ComplaintProcessing, processing_id)
            return p.processing_state if p else "MISSING"


def _record_failure(processing_id: int, err: str) -> None:
    with session_scope() as s:
        p = s.get(ComplaintProcessing, processing_id)
        if p is None:
            return
        p.attempt_count = (p.attempt_count or 0) + 1
        p.last_error = err[:2000]
        if p.attempt_count >= config.MAX_ATTEMPTS:
            p.processing_state = ST_NEEDS_REVIEW
            p.lease_expires_at = None
            log.error("processing_id=%s exhausted retries -> NEEDS_REVIEW", processing_id)
        # else: lease left in place; next tick re-picks after it expires (backoff)
        s.commit()


# ===========================================================================
# State machine
# ===========================================================================

def _run_state_machine(s: Session, p: ComplaintProcessing) -> None:
    c = p.complaint
    instrument = c.instrument if c else None
    incidents = instrument.incidents if instrument else []
    if not instrument or not incidents:
        _to_review(s, p, "no instrument/incidents on complaint")
        return

    # Input checks run once, before any upstream call.
    if p.processing_state == ST_RECEIVED:
        if _ack_only(instrument, incidents):
            _write_response_log(s, p, {"acknowledgement_no": p.acknowledgement_no},
                                response_code=MHA_INVALID_INPUT,
                                response_message="FAILURE", send=False)
            _to_review(s, p, "date and UTR missing: acknowledged only, no callback")
            return
        problem = _input_problem(instrument, incidents)
        if problem:
            _finalize_miscellaneous(s, p, incidents, problem, MHA_INVALID_INPUT)
            _to_review(s, p, problem)
            return

    # ---- Split by amount -------------------------------------------------
    # Incidents at/below the threshold ALWAYS get "Others Less than equal to
    # 500" (txn_type_id 17) with no lookup/hold. Incidents above it follow the
    # normal flow. A complaint may contain both.
    small     = [i for i in incidents if _is_small(i)]
    normal    = [i for i in incidents if not _is_small(i)]
    if small:
        log.info("[%s] %d incident(s) <= %s -> Others Less than equal to 500; "
                 "%d follow the normal flow",
                 p.acknowledgement_no, len(small), config.SMALL_TXN_THRESHOLD,
                 len(normal))
        # Stashed here and appended by _write_response_log, so whichever flow
        # runs for the larger incidents, the small ones ride along in the same
        # response.
        _stash_small_txns(s, p, small)

    if not normal:
        # Everything was small -> respond now, no API calls at all.
        _finalize_small_only(s, p)
        p.processing_state = ST_RESOLVED
        p.lease_expires_at = None
        s.add(p); s.commit()
        return

    # Classify on the NORMAL incidents only, so a small layer-0 incident can't
    # drag the whole complaint's direction/channel.
    cls = classify(instrument, normal)
    account = instrument.payer_account_number
    total_disputed = sum((i.disputed_amount for i in normal), Decimal("0"))

    # Direction is per-incident (layer 0 = always debit). Warn if the normal
    # incidents disagree — the flow below runs on the primary direction.
    dirs = {direction_for_layer(i.layer or 0, instrument.mode_of_payment) for i in normal}
    if len(dirs) > 1:
        log.warning("[%s] mixed directions across layers %s -> using %s",
                    p.acknowledgement_no,
                    sorted((i.layer or 0) for i in normal), cls.direction)

    # From here on the flow operates on `normal` only.
    incidents = normal

    # Channel now comes from the RRN shape. If it matches neither UPI (12-digit)
    # nor IB (16/22-char), do ONE DD transaction enquiry and return whatever it
    # has, rather than forcing the complaint down the wrong flow.
    if cls.needs_dd_probe:
        log.warning("[%s] RRN shape unknown -> DD-enquiry probe",
                    p.acknowledgement_no)
        _finalize_dd_probe(s, p, instrument, incidents)
        p.processing_state = ST_RESOLVED
        p.lease_expires_at = None
        s.add(p); s.commit()
        return

    # Advance one step at a time; commit after each so retries resume here.
    guard = 0
    while p.processing_state not in TERMINAL_STATES and guard < 20:
        guard += 1
        st = p.processing_state
        log.info("[%s] state=%s", p.acknowledgement_no, st)

        if st == ST_RECEIVED:
            p.processing_state = ST_CLASSIFIED

        elif st == ST_CLASSIFIED:
            _enrich(s, p, cls, instrument, incidents)
            p.processing_state = ST_ENRICHED

        elif st == ST_ENRICHED:
            # UPI-debit fraud only needs beneficiary details -> no hold.
            if cls.channel == CHANNEL_UPI and cls.direction == DIR_DEBIT:
                _finalize_bene_only(s, p, incidents)
                p.processing_state = ST_RESOLVED
                p.lease_expires_at = None
                s.add(p); s.commit()
                break
            # IB Debit: DD-validated UTR -> CBDOC bene -> respond, NO hold.
            # If the UTR wasn't validated, return 'miscellaneous' to MHA.
            if cls.direction == DIR_DEBIT:
                if not _rrn_validated(s, p, incidents):
                    _finalize_miscellaneous(s, p, incidents, "Invalid RRN", MHA_INVALID_RRN)
                    p.processing_state = ST_NEEDS_REVIEW
                    p.last_error = "UTR not validated in DD enquiry (miscellaneous)"
                else:
                    _finalize_cbdoc_bene(s, p, incidents)
                    p.processing_state = ST_RESOLVED
                p.lease_expires_at = None
                s.add(p); s.commit()
                break
            # CREDIT: RRN must have validated in the DD enquiry. If not, return a
            # 'miscellaneous' response to MHA and stop — no balance, no hold.
            if cls.direction == DIR_CREDIT and not _rrn_validated(s, p, incidents):
                _finalize_miscellaneous(s, p, incidents, "Invalid RRN", MHA_INVALID_RRN)
                p.processing_state = ST_NEEDS_REVIEW
                p.last_error = "RRN not validated in DD enquiry (miscellaneous)"
                p.lease_expires_at = None
                s.add(p); s.commit()
                break
            p.processing_state = ST_BALANCE_CHECKED

        elif st == ST_BALANCE_CHECKED:
            balance = _available_balance(s, p, account)
            held, remaining = _apply_holds(s, p, instrument, total_disputed, balance)
            # root_effective_balance = balance remaining in the account AFTER the
            # hold (available balance minus the amount we just held).
            post_hold_balance = balance - held
            _stash(s, p, "BALANCE", {"value": str(post_hold_balance)})
            _stash(s, p, "REMAINING", {"value": str(remaining), "held": str(held)})
            p.processing_state = ST_HOLD_APPLIED

        elif st == ST_HOLD_APPLIED:
            remaining = Decimal(_unstash(s, p, "REMAINING").get("value", "0"))
            if remaining <= 0:
                # Full hold covered the disputed amount -> respond now, done.
                _finalize(s, p, instrument, incidents, traced_txns=[])
                p.processing_state = ST_RESOLVED
            else:
                p.processing_state = ST_RECONCILING

        elif st == ST_AWAITING_CBDOC:
            # Parked because a NEFT/RTGS leg needed CBDOC data that wasn't there
            # yet (recent fraud, CBDOC is T+1). Woken next morning -> reconcile
            # again; CBDOC should now have the data.
            p.next_attempt_at = None
            p.processing_state = ST_RECONCILING

        elif st == ST_RECONCILING:
            remaining = Decimal(_unstash(s, p, "REMAINING").get("value", "0"))
            traced, reconciled, cbdoc_pending = _reconcile_credit_from_dd(
                s, p, incidents, remaining)

            # A NEFT/RTGS leg needed CBDOC bene but it wasn't there yet, AND the
            # fraud is recent (CBDOC is T+1). Park and retry next morning at 10am
            # IST. UPI legs never trigger this — they have no 24h restriction.
            if cbdoc_pending and cls.is_recent and p.defer_count < config.MAX_CBDOC_DEFERS:
                wake = _cbdoc_wake_time(incidents)
                p.defer_count += 1
                p.next_attempt_at = wake
                p.processing_state = ST_AWAITING_CBDOC
                p.lease_expires_at = None
                log.info("[%s] NEFT/RTGS bene not in CBDOC yet (recent); retry #%d at %s",
                         p.acknowledgement_no, p.defer_count, wake.isoformat())
                s.add(p); s.commit()
                break

            _finalize(s, p, instrument, incidents, traced_txns=traced)
            p.processing_state = ST_RESOLVED if reconciled else ST_NEEDS_REVIEW
            if not reconciled:
                p.last_error = "debits did not reconcile the disputed amount"
            p.lease_expires_at = None
            s.add(p); s.commit()
            break

        s.add(p)
        s.commit()

    if p.processing_state == ST_RESOLVED:
        p.lease_expires_at = None
        s.commit()


def _to_review(s: Session, p: ComplaintProcessing, reason: str) -> None:
    p.processing_state = ST_NEEDS_REVIEW
    p.last_error = reason
    p.lease_expires_at = None
    s.commit()
    log.error("[%s] -> NEEDS_REVIEW: %s", p.acknowledgement_no, reason)


_ACCOUNT_RE = re.compile(r"[0-9]{12}")   # a string match, so leading zeros survive


def _ack_only(instrument: Instrument, incidents: list[Incident]) -> bool:
    """Account and amount given, but no transaction date and no UTR at all."""
    return bool((instrument.payer_account_number or "").strip()) and all(
        inc.amount and not inc.transaction_date and not (inc.merchant_id or "").strip()
        for inc in incidents)


def _input_problem(instrument: Instrument, incidents: list[Incident]) -> Optional[str]:
    """Why MHA's input can't be processed (-> code 12), or None if it can."""
    if not _ACCOUNT_RE.fullmatch(instrument.payer_account_number or ""):
        return "Invalid account number"
    for inc in incidents:
        if (inc.amount is None or inc.amount <= 0 or inc.disputed_amount is None
                or not inc.transaction_date or not (inc.merchant_id or "").strip()):
            return "Invalid input"
    return None


# ===========================================================================
# Steps (each cached in processing_steps so repeats don't re-hit upstream)
# ===========================================================================

def _step_get(s: Session, p: ComplaintProcessing, step: str,
              idem: Optional[str] = None) -> Optional[ProcessingStep]:
    key = idem or f"{step}:{p.acknowledgement_no}"
    return s.execute(
        select(ProcessingStep).where(ProcessingStep.idempotency_key == key)
    ).scalar_one_or_none()


def _step_record(s: Session, p: ComplaintProcessing, step: str, idem: str,
                 request: dict, response: dict, status: str = "SUCCESS") -> ProcessingStep:
    row = ProcessingStep(
        complaint_id=p.complaint_id, acknowledgement_no=p.acknowledgement_no,
        job_id=p.job_id, step=step, idempotency_key=idem, status=status,
        request_payload=request, response_payload=response,
    )
    try:
        with s.begin_nested():      # a conflict undoes only this insert
            s.add(row)
    except IntegrityError:          # concurrent insert — reuse existing
        return _step_get(s, p, step, idem)   # type: ignore[return-value]
    return row


def _stash(s: Session, p: ComplaintProcessing, name: str, data: dict) -> None:
    idem = f"STASH:{name}:{p.acknowledgement_no}"
    existing = _step_get(s, p, name, idem)
    if existing:
        existing.response_payload = data
        s.flush()
    else:
        _step_record(s, p, f"STASH_{name}", idem, {}, data)


def _unstash(s: Session, p: ComplaintProcessing, name: str) -> dict:
    idem = f"STASH:{name}:{p.acknowledgement_no}"
    row = _step_get(s, p, name, idem)
    return (row.response_payload or {}) if row else {}


def _enrich(s: Session, p: ComplaintProcessing, cls, instrument: Instrument,
            incidents: list[Incident]) -> None:
    """Fetch beneficiary / transaction context.

    Scenarios:
      UPI Debit any age   → UPI API only (no hold)
      CREDIT (any age)    → DD transaction enquiry FIRST to validate the RRN.
                            If RRN not found in any narrative → caller stops and
                            returns a 'miscellaneous' response to MHA (no hold).
                            If found: balance → hold; and if >24h + partial, the
                            reconcile step traces CBDOC debits after the fraud time.
      IB Debit <24h       → DD transaction enquiry → balance → hold
      IB Debit >24h       → CBDOC locate → UPI API if txn is UPI → balance → hold
    """
    account = instrument.payer_account_number

    # --- UPI Debit: short-circuit, bene lookup only, no hold ---
    if cls.channel == CHANNEL_UPI and cls.direction == DIR_DEBIT:
        for inc in incidents:
            idem = f"UPI:{p.acknowledgement_no}:{inc.merchant_id}"
            if _step_get(s, p, "UPI_BENE", idem):
                continue
            ts = f"{inc.transaction_date} {inc.transaction_time}"
            resp = gateway().upi_bene_lookup(
                rrn=inc.merchant_id, account_number=account,
                amount=str(inc.amount), transaction_timestamp=ts,
                ack_id=p.acknowledgement_no,
            )
            _step_record(s, p, "UPI_BENE", idem, {"rrn": inc.merchant_id}, resp)
        return

    # --- CREDIT (any age): validate RRN in the live DD transaction enquiry FIRST ---
    if cls.direction == DIR_CREDIT:
        details = _dd_txn_details(s, p, account, incidents)
        for inc in incidents:
            idem = f"DDTXN:{p.acknowledgement_no}:{inc.merchant_id}"
            if _step_get(s, p, "DD_TXN", idem):
                continue
            matched = _find_txn_by_rrn(details, inc.merchant_id)
            _step_record(s, p, "DD_TXN", idem,
                         {"rrn": inc.merchant_id},
                         {"found": matched is not None, "matched": matched or {}})
            if matched is None:
                log.warning("[%s] RRN=%s not found in DD enquiry narratives",
                            p.acknowledgement_no, inc.merchant_id)
                _log_rrn_miss(p, details, inc.merchant_id)
            else:
                log.info("[%s] RRN=%s MATCHED. transaction=%s",
                         p.acknowledgement_no, inc.merchant_id,
                         json.dumps(matched, default=str))
        return

    # --- IB Debit: validate UTR in DD enquiry, then CBDOC (by UTR) for bene.
    #     No hold — like UPI Debit, but bene comes from CBDOC not the UPI API. ---
    if cls.direction == DIR_DEBIT:
        details = _dd_txn_details(s, p, account, incidents)
        for inc in incidents:
            ddidem = f"DDTXN:{p.acknowledgement_no}:{inc.merchant_id}"
            if not _step_get(s, p, "DD_TXN", ddidem):
                matched = _find_txn_by_rrn(details, inc.merchant_id)
                _step_record(s, p, "DD_TXN", ddidem, {"rrn": inc.merchant_id},
                             {"found": matched is not None, "matched": matched or {}})
                if matched is None:
                    log.warning("[%s] UTR=%s not found in DD enquiry narratives",
                                p.acknowledgement_no, inc.merchant_id)
                    _log_rrn_miss(p, details, inc.merchant_id)
                    continue
                log.info("[%s] UTR=%s MATCHED in DD enquiry", p.acknowledgement_no, inc.merchant_id)

            # UTR validated -> fetch bene details from CBDOC by the UTR/RRN.
            bqidem = f"CBDOCBENE:{p.acknowledgement_no}:{inc.merchant_id}"
            if _step_get(s, p, "CBDOC_BENE", bqidem):
                continue
            bq = gateway().bigquery_search({config.BQ_SEARCH_REF_FIELD: inc.merchant_id}, limit=5)
            _step_record(s, p, "CBDOC_BENE", bqidem, {config.BQ_FIELD_REF: inc.merchant_id}, bq)
            n = len(_rows(bq))
            log.info("[%s] CBDOC bene lookup by UTR=%s -> %d rows",
                     p.acknowledgement_no, inc.merchant_id, n)
        return


def _dd_txn_details(s: Session, p: ComplaintProcessing, account: str,
                    incidents: list[Incident]) -> list[dict]:
    """Run (or reuse) the DD transaction enquiry and return its transaction list."""
    from_date = min(i.transaction_date for i in incidents).isoformat()
    resp = _txn_enquiry(s, p, account, from_date)
    details = _dig(resp, config.TXN_DETAILS_PATH) or []
    if not details:
        # Path yielded nothing — either the real API nests the txn list
        # elsewhere, or the date filter returned no rows. Self-heal: deep-find
        # the list, and log the real shape so the path can be corrected.
        details = _deep_find_txn_list(resp)
        log.warning("[%s] DD enquiry: 0 txns at TXN_DETAILS_PATH='%s'. "
                    "deep-find got %d. response keys=%s",
                    p.acknowledgement_no, config.TXN_DETAILS_PATH,
                    len(details), _shape(resp))
    log.info("[%s] DD enquiry returned %d transactions (fromDate=%s)",
             p.acknowledgement_no, len(details), from_date)
    return details


def _account_call(call):
    """Run an account-level gateway call; HTTP 404 means the account doesn't exist."""
    try:
        return call()
    except GatewayError as e:
        if e.status_code == 404:
            raise AccountNotFound(str(e)) from e
        raise

def _narratives(txn: dict) -> list[str]:
    """Collect every narrative string from a transaction, robust to shape.

    Handles all of:
      transactionDescription: [{transactionNarrative: "..."}, ...]
      transactionDescription: ["...", "..."]            (plain strings)
      transactionNarrative:    "..."  or  ["...", ...]   (directly on txn)
      any nested dict/list containing 'narrative' keys
    """
    out: list[str] = []

    def _add(v):
        if isinstance(v, str) and v.strip():
            out.append(v.strip())

    def _walk(node):
        if isinstance(node, dict):
            for k, v in node.items():
                if "narrative" in k.lower():
                    if isinstance(v, str):
                        _add(v)
                    elif isinstance(v, list):
                        for item in v:
                            _add(item) if isinstance(item, str) else _walk(item)
                    else:
                        _walk(v)
                elif isinstance(v, (dict, list)):
                    _walk(v)
        elif isinstance(node, list):
            for item in node:
                _walk(item)

    # Primary, known shape first (keeps order predictable)
    desc = txn.get("transactionDescription")
    if isinstance(desc, list):
        for d in desc:
            if isinstance(d, dict):
                _add(d.get("transactionNarrative"))
            elif isinstance(d, str):
                _add(d)

    # Then sweep the whole txn for anything missed / differently nested
    _walk(txn)

    # de-dup, preserve order
    seen = set()
    uniq = []
    for n in out:
        if n not in seen:
            seen.add(n)
            uniq.append(n)
    return uniq


def _norm_ref(s: str) -> str:
    """Normalise a reference for comparison: drop all whitespace, uppercase."""
    return re.sub(r"\s+", "", str(s or "")).upper()


def _deep_find_txn_list(resp: Any) -> list[dict]:
    """Find the transaction-details list anywhere in the response, regardless of
    the exact path. A match = a list of dicts where the dicts contain
    'transactionDescription' (or 'transactionNarrative' nested). Lets us recover
    when TXN_DETAILS_PATH doesn't match the real API shape."""
    found: list[dict] = []

    def _looks_like_txn(d: dict) -> bool:
        if not isinstance(d, dict):
            return False
        if "transactionDescription" in d:
            return True
        # some variants put narratives directly on the txn
        return any(k.lower().startswith("transactionnarr") for k in d.keys())

    def _walk(node: Any) -> None:
        nonlocal found
        if found:
            return
        if isinstance(node, list):
            if node and all(isinstance(x, dict) for x in node) and any(_looks_like_txn(x) for x in node):
                found = [x for x in node if isinstance(x, dict)]
                return
            for x in node:
                _walk(x)
        elif isinstance(node, dict):
            for v in node.values():
                _walk(v)

    _walk(resp)
    return found


def _find_txn_by_rrn(details: list[dict], rrn: str) -> Optional[dict]:
    """Find the transaction whose narratives reference this RRN.

    Matching is tolerant: case-insensitive, whitespace-stripped, and matches if
    the RRN equals a narrative, appears as a delimited token (/, space, |, -, :,
    comma), or is contained as a substring (for refs >= 6 chars, to avoid false
    positives on short numbers)."""
    target = _norm_ref(rrn)
    if not target:
        return None
    for txn in details:
        for nv in _narratives(txn):
            nnv = _norm_ref(nv)
            if not nnv:
                continue
            if nnv == target:
                return txn
            # delimited tokens
            tokens = re.split(r"[/\s|,:\-]+", nnv)
            if target in tokens:
                return txn
            # substring (only for sufficiently long refs)
            if len(target) >= 6 and (target in nnv or nnv in target):
                return txn
    return None


def _log_rrn_miss(p: ComplaintProcessing, details: list[dict], rrn: str) -> None:
    """When an RRN doesn't match, dump exactly what we searched vs every
    narrative seen, so a real value mismatch (e.g. 0 vs 8) is visible in logs."""
    seen = []
    for txn in details:
        seen.extend(_narratives(txn))
    log.warning("[%s] RRN miss: looking for '%s' (norm='%s'); narratives seen=%s",
                p.acknowledgement_no, rrn, _norm_ref(rrn), seen[:40])

def _available_balance(s: Session, p: ComplaintProcessing, account: str) -> Decimal:
    """Balance used for the hold decision. Tries the configured path first;
    if that misses, deep-searches for BALANCE_FIELD (accepting either a scalar
    or an {amount} object) and logs the real path. Raises with the response
    SHAPE (keys only) if nothing usable is found."""
    resp = _dd_enquiry(s, p, account)

    bal = _coerce_balance(_dig(resp, config.BALANCE_PATH))
    if bal is None:
        found, found_path = _deep_find_balance(resp, config.BALANCE_FIELD)
        bal = _coerce_balance(found)
        if bal is not None:
            log.warning(
                "[%s] balance not at BALANCE_PATH='%s'; found via deep search at "
                "'%s' — set BALANCE_PATH='%s' to make it explicit.",
                p.acknowledgement_no, config.BALANCE_PATH, found_path, found_path,
            )
    if bal is None:
        raise GatewayError(
            f"balance not found. tried BALANCE_PATH='{config.BALANCE_PATH}' and "
            f"deep search for '{config.BALANCE_FIELD}'. "
            f"response shape (keys only) = {_shape(resp)}"
        )
    _stash(s, p, "BALANCE", {"value": str(bal)})   # for root_effective_balance
    return bal

def _apply_holds(s: Session, p: ComplaintProcessing, instrument: Instrument,
                 disputed: Decimal, balance: Decimal) -> tuple[Decimal, Decimal]:
    """Hold min(disputed, balance). Returns (amount_held, remaining_to_trace)."""
    to_hold = min(disputed, balance) if balance > 0 else Decimal("0")
    remaining = disputed - to_hold

    if to_hold <= 0:
        log.info("[%s] no balance to hold; remaining=%s", p.acknowledgement_no, remaining)
        return Decimal("0"), remaining

    account = instrument.payer_account_number
    currency = config.HOLD_CURRENCY
    amount_str = f"{to_hold:.2f}"
    # Idempotency key ties the hold to (ack, account, amount): a retry that
    # finds a SUCCESS here will NOT call apply-hold again.
    idem = f"HOLD:{p.acknowledgement_no}:{account}:{amount_str}"

    existing = _step_get(s, p, "HOLD", idem)
    if existing and existing.status == "SUCCESS":
        log.info("[%s] hold already applied (idempotent skip)", p.acknowledgement_no)
        return to_hold, remaining

    # Reserve the key first (PENDING). If another worker raced us, the UNIQUE
    # constraint makes _step_record return the existing row instead.
    step = _step_record(s, p, "HOLD", idem,
                        {"account": account, "amount": amount_str},
                        {}, status="PENDING")
    if step.status == "SUCCESS":
        return to_hold, remaining

    resp = gateway().apply_hold(
        account_number=account, currency=currency, amount=amount_str,
        hold_type=config.HOLD_TYPE, hold_till=config.HOLD_TILL_DATE,
        hold_narrative2=f"FRAUD {p.acknowledgement_no}"[:35],
        hold_narrative1=f"AckNo.-{p.acknowledgement_no}-MHA",
    )
    step.status = "SUCCESS"
    step.response_payload = resp
    s.flush()
    log.info("[%s] hold applied amount=%s remaining=%s",
             p.acknowledgement_no, amount_str, remaining)
    return to_hold, remaining

def _txn_amount(txn: dict) -> Decimal:
    """Signed transaction amount from a DD-enquiry txn. Debit is negative."""
    a = _dig(txn, "transactionAmount.amount")
    try:
        return Decimal(str(a)) if a not in (None, "") else Decimal("0")
    except Exception:
        return Decimal("0")


def _is_debit_txn(txn: dict) -> bool:
    """A debit = money leaving the account = negative transactionAmount."""
    return _txn_amount(txn) < 0


def _txn_time(txn: dict) -> datetime:
    """Transaction time (tz-aware UTC) from executionTimestamp, else value/post
    date. datetime.min(UTC) if nothing usable."""
    ts = _dig(txn, "transactionId.executionTimestamp")
    if ts:
        try:
            dt = datetime.fromisoformat(str(ts))
            dt = dt.replace(tzinfo=timezone.utc) if dt.tzinfo is None else dt
            return dt.astimezone(timezone.utc)
        except Exception:
            pass
    for k in ("valueDate", "postDate"):
        d = txn.get(k)
        if d:
            try:
                return datetime.fromisoformat(str(d)).replace(tzinfo=timezone.utc)
            except Exception:
                pass
    return datetime.min.replace(tzinfo=timezone.utc)


def _fmt_txn_time(txn: dict) -> str:
    dt = _txn_time(txn)
    m = datetime.min.replace(tzinfo=timezone.utc)
    return dt.strftime("%Y-%m-%d %H:%M:%S") if dt != m else ""


def _scheme_from_narratives(txn: dict) -> Optional[str]:
    """Detect the payment scheme from a transaction's narratives.
    A scheme tag may be a bare token ('UPI', 'IMPS') OR a prefixed string
    like 'UPI2026063000015887a'. The tag is separate from the numeric RRN."""
    for nv in _narratives(txn):
        u = nv.strip().upper()
        for tok in re.split(r"[/\s,;|]+", u):
            if tok == "ATM":        # exact only, so names like "ATMARAM" don't match
                return "ATM"
            for scheme in ("IMPS", "UPI", "NEFT", "RTGS"):
                # bare token, or a token that starts with the scheme (e.g. UPI2026...)
                if tok == scheme or tok.startswith(scheme):
                    return scheme
    return None


def _classify_rrn(token: str, payer_digits: str,
                  scheme: Optional[str] = None) -> Optional[tuple[str, str]]:
    """Classify a narrative token as an RRN. Returns (rrn, type) or None.
      HSBCN...  -> NEFT
      HSBCR...  -> RTGS
      12 digits -> use the transaction's scheme tag if present (UPI vs IMPS),
                   else UPI (the UPI-API probe later confirms/denies).
      payer's own 12-digit account -> not an RRN.
    """
    t = (token or "").strip()
    if not t:
        return None
    up = t.upper()
    # NEFT/RTGS UTRs are 22 chars (e.g. HSBCN52026060900034931). Shorter HSBCN*/
    # HSBCR* tokens (e.g. HSBCN16000034931, 16 chars) are internal refs, NOT the
    # UTR — reject them so we don't create a duplicate transaction.
    if up.startswith("HSBCN"):
        return (t, "NEFT") if len(t) == config.UTR_LENGTH else None
    if up.startswith("HSBCR"):
        return (t, "RTGS") if len(t) == config.UTR_LENGTH else None
    if t.isdigit() and len(t) == 12:
        if payer_digits and t in payer_digits:      # payer's own account, not an RRN
            return None
        if scheme in ("UPI", "IMPS", "ATM"):
            return (t, scheme)                        # scheme tag decides
        return (t, "UPI")                             # no tag -> assume UPI, probe confirms
    return None


def _extract_rrns(txn: dict, payer_digits: str) -> list[tuple[str, str]]:
    """All classifiable RRNs from a txn's narratives, scheme-aware, de-duplicated."""
    scheme = _scheme_from_narratives(txn)
    out: dict[str, str] = {}
    for nv in _narratives(txn):
        for tok in re.split(r"[/\s,;|]+", nv):
            c = _classify_rrn(tok, payer_digits, scheme)
            if c and c[0] not in out:
                out[c[0]] = c[1]
    return list(out.items())


def _bene_for_rrn(s: Session, p: ComplaintProcessing, rrn: str, rtype: str,
                  account: str, amount: Decimal, ts_iso: str) -> tuple[dict, str, str]:
    """Fetch bene details for one RRN. Returns (bene, final_type, status).

      IMPS/ATM  -> out of scope, no lookup. status='ops_review' (MHA gets 31 Pending).
      UPI       -> UPI API. Valid -> status='resolved'. Invalid response means it was
                   actually IMPS -> downgrade type to IMPS, status='ops_review'.
      NEFT/RTGS -> CBDOC by LPUTRN. status='resolved' (empty row -> still resolved,
                   just blank bene).
    status is one of: 'resolved' | 'ops_review'.
    """
    if rtype in ("IMPS", "ATM"):
        return {}, rtype, "ops_review"

    if rtype == "UPI":
        idem = f"UPI:{p.acknowledgement_no}:{rrn}"
        step = _step_get(s, p, "UPI_BENE", idem)
        if not step:
            try:
                resp = gateway().upi_bene_lookup(
                    rrn=rrn, account_number=account, amount=str(amount),
                    transaction_timestamp=ts_iso, ack_id=p.acknowledgement_no)
                step = _step_record(s, p, "UPI_BENE", idem, {"rrn": rrn}, resp)
            except GatewayError as e:
                # UPI API rejected it -> not a UPI RRN -> treat as IMPS (ops).
                log.info("[%s] UPI API invalid for rrn=%s (%s) -> IMPS/ops",
                         p.acknowledgement_no, rrn, e)
                _step_record(s, p, "UPI_BENE", idem, {"rrn": rrn},
                             {"invalid": True, "error": str(e)}, status="FAILURE")
                return {}, "IMPS", "ops_review"
        data = _dig(step.response_payload, "data") or step.response_payload or {}
        if not data or (step.response_payload or {}).get("invalid"):
            # cached invalid, or empty payload -> IMPS/ops
            return {}, "IMPS", "ops_review"
        return _normalise_bene(data), "UPI", "resolved"

    # NEFT / RTGS -> CBDOC by LPUTRN. Include defer_count so each morning's
    # retry re-queries CBDOC instead of reusing a cached empty result.
    idem = f"CBDOCBENE:{p.acknowledgement_no}:{rrn}:{p.defer_count}"
    step = _step_get(s, p, "CBDOC_BENE", idem)
    if not step:
        try:
            resp = gateway().bigquery_search({config.BQ_SEARCH_REF_FIELD: rrn}, limit=5)
            step = _step_record(s, p, "CBDOC_BENE", idem, {config.BQ_SEARCH_REF_FIELD: rrn}, resp)
        except GatewayError as e:
            log.warning("[%s] CBDOC bene failed rrn=%s: %s", p.acknowledgement_no, rrn, e)
            return {}, rtype, "resolved"
    rows = _rows(step.response_payload)
    if not rows:
        log.warning("[%s] CBDOC bene: 0 rows searching %s=%s (resp keys=%s)",
                    p.acknowledgement_no, config.BQ_SEARCH_REF_FIELD, rrn,
                    list(step.response_payload.keys()) if isinstance(step.response_payload, dict) else type(step.response_payload).__name__)
        # No CBDOC row yet — caller may retry (CBDOC is T+1 for recent frauds).
        return {}, rtype, "cbdoc_empty"
    row = _cbdoc_row(rows, rrn)
    if row is None:
        log.warning("[%s] CBDOC bene: %d row(s) for %s, none dated like the UTR",
                    p.acknowledgement_no, len(rows), rrn)
        return {}, rtype, "invalid_rrn"
    # Beneficiary name = OGBEA1..OGBEA4 lines joined; account = OGBEAC (strip '/').
    name = _bene_name(row)
    bene = {
        "account": _strip_acct(row.get(config.BQ_FIELD_BENE_ACCOUNT, "")),   # OGBEAC
        "bank": "",                                                          # no bank column in CBDOC
        "name": name,                                                        # OGBEA1..OGBEA4
        "ifsc": str(row.get(config.BQ_FIELD_IFSC, "") or "").strip(),
        "bankCode": "",
        "vpa": "",
        "currency": row.get("OGPYCY", ""),
        "txn_datetime": _fmt_row_dt(row),          # OGTMZ1 -> response transaction_datetime
        "narratives": [row.get(k) for k in ("OGNAR1", "OGNAR2", "OGNAR3", "OGNAR4") if row.get(k)],
    }
    log.info("[%s] CBDOC bene rrn=%s -> account=%s name=%s",
             p.acknowledgement_no, rrn, bene["account"], bene["name"][:40])
    return bene, rtype, "resolved"

# MHA REJECTS the request if any unexpected field is present, so these three

# ===========================================================================

def _leg_bank_code(bene: dict, txn: dict) -> str:
    """Bank code for a traced debit leg.

    UPI legs carry an IFSC / bank name / VPA, so those resolve directly.
    NEFT/RTGS legs come from CBDOC with NO IFSC, so fall back to scanning the
    DD transaction's narratives, where the bank is usually named.
    """
    code = bank_codes.resolve_bank_code(
        ifsc=bene.get("ifsc", ""), bank_name=bene.get("bank", ""))
    if code != config.MHA_DEFAULT_BANK_CODE:
        return code

    code, matched = bank_codes.resolve_from_narratives(_narratives(txn))
    if matched:
        log.info("leg bank resolved from DD narrative %r -> code=%s", matched, code)
    return code


def _mha_debit_obj(rrn: str, rtype: str, amount: Decimal, txn: dict,
                   bene: dict, root: dict, disputed: Decimal,
                   bene_status: str = "resolved") -> dict:
    """One traced debit leg -> MHA 'Money Transfer To' object."""
    code, remarks = _LEG_STATUS.get(bene_status, (MHA_OK, ""))
    return _mha_money_transfer_to(
        rrn=rrn,
        payee_bank=bene.get("bank", ""),
        payee_bank_code=_leg_bank_code(bene, txn),
        payee_account=bene.get("account", ""),
        ifsc=bene.get("ifsc", ""),
        amount=str(amount),
        txn_datetime=bene.get("txn_datetime") or _fmt_txn_time(txn),
        disputed=str(disputed),
        root_account=root["root_account_number"],
        root_bankid=root["root_bankid"],
        effective_balance=root["root_effective_balance"],
        status_code=code,
        remarks=remarks,
    )


def _min_incident_utc(incidents: list[Incident]) -> datetime:
    fraud_ist = min((_incident_dt(inc) for inc in incidents), default=datetime.min)
    return fraud_ist.replace(tzinfo=_IST).astimezone(timezone.utc)


def _reconcile_credit_from_dd(s: Session, p: ComplaintProcessing,
                              incidents: list[Incident],
                              remaining: Decimal) -> tuple[list[dict], bool, bool]:
    """Reconcile a CREDIT fraud from the DD transaction enquiry.

      1. Reuse the DD enquiry transaction list.
      2. Anchor on the credit (fraud RRN) transaction time.
      3. Walk DEBITS (negative amount) after the credit, chronologically,
         accumulating |amount| until `remaining` is reconciled.
      4. From each debit's narratives, extract RRNs and classify NEFT/RTGS/UPI.
      5. UPI -> UPI API bene; NEFT/RTGS -> CBDOC(LPUTRN) bene.
      6. Return the collated MHA objects + reconciled flag."""
    inst = p.complaint.instrument
    account = inst.payer_account_number
    payer_digits = re.sub(r"\D", "", account or "")
    from_date = min(i.transaction_date for i in incidents).isoformat()

    resp = _txn_enquiry(s, p, account, from_date)
    details = _dig(resp, config.TXN_DETAILS_PATH) or _deep_find_txn_list(resp)

    fraud_refs = [inc.merchant_id for inc in incidents]
    credit = None
    for ref in fraud_refs:
        credit = _find_txn_by_rrn(details, ref)
        if credit:
            break
    anchor = _txn_time(credit) if credit else _min_incident_utc(incidents)
    log.info("[%s] DD reconcile: %d txns, anchor=%s (credit %sfound)",
             p.acknowledgement_no, len(details), anchor.isoformat(),
             "" if credit else "NOT ")

    debits = sorted((t for t in details if _is_debit_txn(t)), key=_txn_time)
    objs: list[dict] = []
    recovered = Decimal("0")
    ops_pending = 0
    root = _root_ctx(s, p)
    disputed = sum((i.disputed_amount for i in incidents), Decimal("0"))
    seen: set[str] = set()
    cbdoc_pending = False

    for txn in debits:
        if _txn_time(txn) < anchor:          # only debits at/after the credit
            continue
        amt = abs(_txn_amount(txn))
        if amt <= 0:
            continue
        ts_iso = _fmt_txn_time(txn)
        for rrn, rtype in _extract_rrns(txn, payer_digits):
            if rrn in seen:
                continue
            seen.add(rrn)
            bene, final_type, status = _bene_for_rrn(s, p, rrn, rtype, account, amt, ts_iso)
            if status == "cbdoc_empty":
                # NEFT/RTGS bene not in CBDOC yet -> signal a retry. Emit the leg
                # as pending so the amount still counts toward reconciliation.
                cbdoc_pending = True
                objs.append(_mha_debit_obj(rrn, final_type, amt, txn, bene, root, disputed,
                                           bene_status="cbdoc_pending"))
            else:
                objs.append(_mha_debit_obj(rrn, final_type, amt, txn, bene, root, disputed,
                                           bene_status=status))
                if status == "ops_review":
                    ops_pending += 1
            log.info("[%s] traced debit rrn=%s type=%s status=%s amount=%s",
                     p.acknowledgement_no, rrn, final_type, status, amt)
        recovered += amt                      # IMPS included: the money DID leave
        if recovered >= remaining:
            break

    fully = recovered >= remaining
    log.info("[%s] DD reconcile done: objs=%d ops_pending=%d cbdoc_pending=%s "
             "recovered=%s remaining=%s reconciled=%s",
             p.acknowledgement_no, len(objs), ops_pending, cbdoc_pending,
             recovered, remaining, fully)
    return objs, fully, cbdoc_pending


def _rows(bq_resp: Any) -> list[dict]:
    """Extract the CBDOC row list, tolerant of response shape:
      {"rows":[...]}            (gateway direct)
      {"data":{"rows":[...]}}   (data-enveloped)
      [ ... ]                   (bare list)
    """
    rows: Any = []
    if isinstance(bq_resp, list):
        rows = bq_resp
    elif isinstance(bq_resp, dict):
        data = bq_resp.get("data")
        if isinstance(bq_resp.get("rows"), list):
            rows = bq_resp["rows"]
        elif isinstance(data, dict) and isinstance(data.get("rows"), list):
            rows = data["rows"]
        elif isinstance(data, list):
            rows = data
    return [r for r in rows if isinstance(r, dict) and r]   # junk/empty rows never match


def _parse_ogtmz1(v: Any) -> Optional[datetime]:
    """Parse CBDOC OGTMZ1 'YYYY-MM-DD-HH.MM.SS.ffffff' (IST) into a datetime.
    Tolerates missing microseconds. Returns None if unparseable."""
    if not v:
        return None
    s = str(v).strip()
    for fmt in ("%Y-%m-%d-%H.%M.%S.%f", "%Y-%m-%d-%H.%M.%S"):
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    return None


def _utr_date(utr: str) -> Optional[str]:
    """yyyyMMdd embedded in a NEFT/RTGS UTR: HSBCN5 20260609 00034931 -> '20260609'.
    None for any other shape (16-char UTRs, internal refs, junk)."""
    t = (utr or "").strip().upper()
    if len(t) != config.UTR_LENGTH or not t.startswith(("HSBCN", "HSBCR")):
        return None
    d = t[6:14]
    try:
        datetime.strptime(d, "%Y%m%d")
    except ValueError:
        return None
    return d if d.isdigit() else None


def _cbdoc_row(rows: list[dict], utr: str) -> Optional[dict]:
    """The one CBDOC row for this UTR, or None if no row's OGTMZ1 date matches
    the UTR's embedded date. Identical rows collapse; distinct ones raise."""
    want = _utr_date(utr)
    if want:
        rows = [r for r in rows
                if (ts := _parse_ogtmz1(r.get(config.BQ_FIELD_TIMESTAMP)))
                and ts.strftime("%Y%m%d") == want]
    unique = list({json.dumps(r, sort_keys=True, default=str): r for r in rows}.values())
    if len(unique) > 1:
        raise CbdocConflict(f"CBDOC returned {len(unique)} different rows for UTR {utr}")
    return unique[0] if unique else None


def _row_dt(row: dict) -> datetime:
    """Real transaction time of a CBDOC row. Prefers OGTMZ1; falls back to
    OGCPDT (YYYYMMDD) + OGCPTM (HHMMSS). datetime.min if nothing usable."""
    ts = _parse_ogtmz1(row.get(config.BQ_FIELD_TIMESTAMP))
    if ts is not None:
        return ts
    def _int(x):
        try:
            return int(x)
        except Exception:
            return 0
    d, t = _int(row.get(config.BQ_FIELD_DATE)), _int(row.get(config.BQ_FIELD_TIME))
    if not d:
        return datetime.min
    ts6 = str(t).zfill(6)
    try:
        return datetime(d // 10000, (d // 100) % 100, d % 100,
                        int(ts6[0:2]), int(ts6[2:4]), int(ts6[4:6]))
    except Exception:
        return datetime.min


_IST = timezone(timedelta(hours=5, minutes=30))


def _cbdoc_wake_time(incidents: list[Incident]) -> datetime:
    """Next CBDOC retry time as tz-aware UTC.

    CBDOC updates overnight (T+1), so recent (<24h) cases that need CBDOC are
    retried the NEXT MORNING at CBDOC_RETRY_HOUR_IST (default 10:00 IST). If it's
    already past that hour today, wake tomorrow; otherwise wake today at that hour.
    """
    now_ist = _now().astimezone(_IST)
    target = now_ist.replace(hour=config.CBDOC_RETRY_HOUR_IST, minute=0,
                             second=0, microsecond=0)
    if now_ist >= target:
        target = target + timedelta(days=1)
    return target.astimezone(timezone.utc)


def _incident_dt(inc: Incident) -> datetime:
    """Fraud time from the MHA request as a datetime (IST; no conversion needed,
    OGTMZ1 is already IST). Used to collect only CBDOC txns AFTER the fraud time."""
    d, t = inc.transaction_date, inc.transaction_time
    if not d:
        return datetime.min
    return datetime(d.year, d.month, d.day,
                    t.hour if t else 0, t.minute if t else 0, t.second if t else 0)


def _bene_name(row: dict) -> str:
    """Beneficiary name = CBDOC OGBEA1..OGBEA4 lines joined."""
    return " ".join(
        str(row.get(k, "")).strip() for k in ("OGBEA1", "OGBEA2", "OGBEA3", "OGBEA4")
        if str(row.get(k, "")).strip()
    )


def _strip_acct(v: Any) -> str:
    """Strip the leading '/' CBDOC stores on account values."""
    s = str(v or "").strip()
    return s[1:] if s.startswith("/") else s


def _fmt_row_dt(row: dict) -> str:
    """Format a CBDOC row's real transaction time for the MHA response."""
    dt = _row_dt(row)
    return dt.strftime("%Y-%m-%d %H:%M:%S") if dt != datetime.min else ""


def _root_ctx(s: Session, p: ComplaintProcessing) -> dict:
    """The 'root' = the original fraud account + first incident's RRN + the
    account's balance. Shared across every response object."""
    inst = p.complaint.instrument
    incs = inst.incidents if inst else []
    primary = incs[0] if incs else None
    return {
        "root_account_number": inst.payer_account_number if inst else "",
        "root_rrn_transaction_id": (primary.merchant_id if primary else ""),
        "root_bankid": (inst.payer_bank_code if inst else "") or "",
        "root_effective_balance": _unstash(s, p, "BALANCE").get("value", ""),
        "root_ifsc_code": "",          # TODO: source IFSC of the root account
    }


# Fields we don't yet have a source for are emitted as "" so the JSON shape is
# complete. Wire them up once you confirm where they come from (transaction
# enquiry narratives, a customer-info service, or extra CBDOC columns).
def _find_key(obj: Any, key: str) -> Any:
    """Recursively find the first occurrence of `key` anywhere in a nested
    dict/list structure. Returns None if absent. Used because the decrypted UPI
    payload nesting varies between HSBC environments."""
    if isinstance(obj, dict):
        if key in obj and obj[key] not in (None, ""):
            return obj[key]
        for v in obj.values():
            found = _find_key(v, key)
            if found is not None:
                return found
    elif isinstance(obj, list):
        for item in obj:
            found = _find_key(item, key)
            if found is not None:
                return found
    return None


def _normalise_bene(data: dict) -> dict:
    """Map the real UPI API response to a canonical bene dict.

    UPI response keys (the BENEFICIARY — where the money went):
      pyAccNo     -> beneficiary account  == MHA payee_account_number
      pyIfscCode  -> beneficiary IFSC
      bankName    -> beneficiary bank (short name)
      bankCode    -> beneficiary bank code
      pyFname     -> beneficiary name
      pyFvaddr    -> beneficiary VPA

    NOTE: the account we SEND in the request (`accNo`) is the payer/root
    account and maps to MHA root_account_number — not to payee_account_number.

    Keys are deep-searched so extra nesting in the decrypted payload doesn't
    silently blank the fields.
    """
    data = data or {}

    def g(key: str) -> str:
        v = data.get(key)
        if v in (None, ""):
            v = _find_key(data, key)      # fall back to a deep search
        return str(v) if v not in (None, "") else ""

    bene = {
        "account":  g("pyAccNo"),          # -> payee_account_number
        "ifsc":     g("pyIfscCode"),       # -> ifsc_code
        "bank":     g("bankName"),         # -> payee_bank
        # NOTE: UPI's `bankCode` is NOT mapped. In real responses it repeats the
        # IFSC (e.g. bankCode == pyIfscCode == "HSBC0400002"), so feeding it into
        # payee_bank_code polluted that field (and root_bankid) with an IFSC.
        "bankCode": "",
        "name":     g("pyFname"),
        "vpa":      g("pyFvaddr"),
        "txn_datetime": _fmt_upi_date(g("txnDate")),   # -> transaction_datetime
    }
    if not bene["account"]:
        log.warning("UPI bene: pyAccNo not found. payload keys=%s",
                    list(data.keys()) if isinstance(data, dict) else type(data).__name__)
    else:
        log.info("UPI bene -> acct=%s bank=%s ifsc=%s txnDate=%s",
                 bene["account"], bene["bank"], bene["ifsc"], bene["txn_datetime"])
    return bene


def _fmt_upi_date(raw: str) -> str:
    """UPI txnDate 'YYYY/MM/DD HH:MM:SS' -> MHA 'YYYY-MM-DD HH:MM:SS'."""
    t = (raw or "").strip()
    if not t:
        return ""
    for fmt in ("%Y/%m/%d %H:%M:%S", "%Y-%m-%d %H:%M:%S",
                "%d-%m-%Y %H:%M:%S", "%Y/%m/%d"):
        try:
            return datetime.strptime(t, fmt).strftime("%Y-%m-%d %H:%M:%S")
        except ValueError:
            continue
    log.warning("UPI txnDate unparsed: %r", raw)
    return t.replace("/", "-")          # best effort

def _hold_txns(s: Session, p: ComplaintProcessing, root: dict) -> list[dict]:
    """Applied holds -> MHA 'Transaction Put on Hold' objects (txn_type_id 1).
    One object per hold."""
    rows = s.execute(
        select(ProcessingStep).where(
            ProcessingStep.complaint_id == p.complaint_id,
            ProcessingStep.step == "HOLD",
            ProcessingStep.status == "SUCCESS",
        )
    ).scalars().all()
    out = []
    for r in rows:
        req = r.request_payload or {}
        out.append(_mha_put_on_hold(
            account=req.get("account", root["root_account_number"]),
            rrn=root["root_rrn_transaction_id"],
            bankid=root["root_bankid"],
            amount=req.get("amount", ""),
            hold_datetime=_fmt_hold_dt(r.created_at),   # when the hold was placed
            effective_balance=root["root_effective_balance"],
        ))
    return out


def _fmt_hold_dt(dt: Optional[datetime]) -> str:
    """Format a hold's placement time as 'YYYY-MM-DD HH:MM:SS' in IST."""
    if not dt:
        return ""
    if dt.tzinfo is None:                     # naive -> assume UTC
        dt = dt.replace(tzinfo=timezone.utc)
    return dt.astimezone(_IST).strftime("%Y-%m-%d %H:%M:%S")


def _finalize(s: Session, p: ComplaintProcessing, instrument: Instrument,
              incidents: list[Incident], traced_txns: list[dict]) -> None:
    """transactions = one 'Transaction Put on Hold' per hold + one 'Money
    Transfer To' per traced CBDOC debit (already built with root context)."""
    root = _root_ctx(s, p)
    payload = {
        "acknowledgement_no": p.acknowledgement_no,
        "job_id": p.job_id,
        "transactions": _hold_txns(s, p, root) + traced_txns,
    }
    _write_response_log(s, p, payload)


def _finalize_cbdoc_bene(s: Session, p: ComplaintProcessing, incidents: list[Incident]) -> None:
    """IB Debit: bene details come from the CBDOC row matched by the UTR/RRN.
    One 'Money Transfer To' object per incident. No hold."""
    root = _root_ctx(s, p)
    txns = []
    for inc in incidents:
        step = _step_get(s, p, "CBDOC_BENE", f"CBDOCBENE:{p.acknowledgement_no}:{inc.merchant_id}")
        rows = _rows(step.response_payload) if step and step.response_payload else []
        row = _cbdoc_row(rows, inc.merchant_id) if rows else {}
        if row is None:
            txns.append(_incident_failure(root, inc, MHA_INVALID_RRN, "Invalid RRN"))
            continue
        name = _bene_name(row)
        txns.append(_mha_money_transfer_to(
            rrn=inc.merchant_id,
            payee_bank=name,                 # CBDOC OGBEA1-4 = beneficiary name
            payee_bank_code=bank_codes.resolve_bank_code(narrative=name),
            payee_account=_strip_acct(row.get(config.BQ_FIELD_BENE_ACCOUNT, "")),
            amount=str(inc.amount),
            txn_datetime=_fmt_row_dt(row)
                         or f"{inc.transaction_date} {inc.transaction_time}",
            disputed=str(inc.disputed_amount),
            root_account=root["root_account_number"],
        root_bankid=root["root_bankid"],
            effective_balance=root["root_effective_balance"],
            remarks=row.get("OGNAR1", "") or "",
        ))
    _write_response_log(s, p, {
        "acknowledgement_no": p.acknowledgement_no,
        "job_id": p.job_id,
        "transactions": txns,
    })

    }


def _is_small(inc: Incident) -> bool:
    """True if this incident is at/below the small-transaction threshold."""
    return (inc.amount or Decimal("0")) <= config.SMALL_TXN_THRESHOLD

def _finalize_small_only(s: Session, p: ComplaintProcessing) -> None:

    })


def _finalize_dd_probe(s: Session, p: ComplaintProcessing, instrument: Instrument,
                       incidents: list[Incident]) -> None:
    """Fallback for RRNs whose shape matches neither UPI (12-digit) nor IB
    (16/22-char). Runs ONE DD transaction enquiry; if the RRN is found, returns
    that transaction's data. Otherwise responds 'Invalid RRN'."""
    account = instrument.payer_account_number
    from_date = min(i.transaction_date for i in incidents).isoformat()
    root = _root_ctx(s, p)

    try:
        resp = _txn_enquiry(s, p, account, from_date)
    except GatewayError as e:
        log.error("[%s] DD probe failed: %s", p.acknowledgement_no, e)
        _finalize_miscellaneous(s, p, incidents, f"DD enquiry failed: {e}",
                                MHA_RECORD_NOT_FOUND)
        return

    details = _dig(resp, config.TXN_DETAILS_PATH) or _deep_find_txn_list(resp)
    txns: list[dict] = []
    for inc in incidents:
        found = _find_txn_by_rrn(details, inc.merchant_id)
        if not found:
            log.info("[%s] DD probe: rrn=%s not found", p.acknowledgement_no, inc.merchant_id)
            txns.append(_incident_failure(root, inc, MHA_INVALID_RRN, "Invalid RRN"))
            continue

        log.info("[%s] DD probe: rrn=%s FOUND in DD enquiry",
                 p.acknowledgement_no, inc.merchant_id)
        amt = abs(_txn_amount(found)) or inc.amount
        txns.append(_mha_money_transfer_to(
            rrn=inc.merchant_id,
            payee_bank="", payee_bank_code="", payee_account="",
            amount=str(amt),
            txn_datetime=_fmt_txn_time(found)
                         or f"{inc.transaction_date} {inc.transaction_time}",
            disputed=str(inc.disputed_amount),
            root_account=account,
            root_bankid=root["root_bankid"],
            effective_balance=root["root_effective_balance"],
            remarks="Retrieved from transaction enquiry"))

    _write_response_log(s, p, {
        "acknowledgement_no": p.acknowledgement_no,
        "job_id": p.job_id,
        "transactions": txns,
    })


def _finalize_bene_only(s: Session, p: ComplaintProcessing, incidents: list[Incident]) -> None:
    """UPI-debit: one 'Money Transfer To' per incident, populated from the UPI
    beneficiary lookup."""
    root = _root_ctx(s, p)
    txns = []
    for inc in incidents:
        step = _step_get(s, p, "UPI_BENE", f"UPI:{p.acknowledgement_no}:{inc.merchant_id}")
        raw = (_dig(step.response_payload, "data") or step.response_payload) if step and step.response_payload else {}
        bene = _normalise_bene(raw or {})
        txns.append(_mha_money_transfer_to(
            rrn=inc.merchant_id,
            payee_bank=bene.get("bank", ""),
            payee_bank_code=bank_codes.resolve_bank_code(
                ifsc=bene.get("ifsc", ""), bank_name=bene.get("bank", "")),
            payee_account=bene.get("account", ""),
            ifsc=bene.get("ifsc", ""),
            amount=str(inc.amount),
            txn_datetime=bene.get("txn_datetime")
                         or f"{inc.transaction_date} {inc.transaction_time}",
            disputed=str(inc.disputed_amount),
            root_account=root["root_account_number"],
        root_bankid=root["root_bankid"],
            effective_balance=root["root_effective_balance"],
        ))
    _write_response_log(s, p, {
        "acknowledgement_no": p.acknowledgement_no,
        "job_id": p.job_id,
        "transactions": txns,
    })


def _write_response_log(s: Session, p: ComplaintProcessing, payload: dict,
                        response_code: str = "00",
                        response_message: str = "SUCCESS", send: bool = True) -> None:
    # Any <=500 incidents ride along in the same response as the larger ones.
    small = (_unstash(s, p, "SMALL_TXNS") or {}).get("objs") or []
    if small:
        payload = dict(payload)
        payload["transactions"] = list(payload.get("transactions", [])) + small

    existing = s.execute(
        select(ApiResponseLog).where(ApiResponseLog.job_id == p.job_id)
    ).scalar_one_or_none()
    if existing:
        existing.response_payload = payload
        existing.response_code = response_code
        existing.response_message = response_message
        existing.responded_at = _now()
    else:
        s.add(ApiResponseLog(
            complaint_id=p.complaint_id, acknowledgement_no=p.acknowledgement_no,
            job_id=p.job_id, response_code=response_code, response_message=response_message,
            response_payload=payload, responded_at=_now(),
        ))
    s.flush()
    log.info("[%s] response log written job_id=%s txns=%d",
             p.acknowledgement_no, p.job_id, len(payload.get("transactions", [])))

    # Encrypt and send the final response to MHA (via the gateway). Every
    # finalizer funnels through here, so all paths send consistently.
    if send:
        _send_mha_response(s, p, payload)


_JOB_ID_RE = re.compile(r"BANKS-[0-9a-fA-F-]{8,}")


def _extract_encrypted_ack(res: Any) -> str:
    """Pull the encrypted base64 ack string out of the gateway's reply.

    The gateway wraps the upstream body as {"requestId":..., "data": <body>},
    and because MHA replies with a bare text/plain string, `data` is usually
    {"raw": "<base64>"}. Tolerate the other shapes too.
    """
    if isinstance(res, str):
        return res
    if not isinstance(res, dict):
        return ""
    data = res.get("data", res)
    if isinstance(data, str):
        return data
    if isinstance(data, dict):
        for k in ("raw", "response", "body", "data", "responseMsg"):
            v = data.get(k)
            if isinstance(v, str) and v.strip():
                return v
    return ""


def _store_mha_ack(s: Session, p: ComplaintProcessing, res: Any) -> None:
    """Decrypt MHA's acknowledgement and store it on the matching
    api_response_logs row.

    The ack's `message` carries our job_id, e.g.
      "Response has been successfully added. BANKS-8996cda1-..."
    That job_id is the identifier used to find the row. Entirely non-fatal —
    a failure here must never undo a successfully-sent response.
    """
    encrypted_ack = _extract_encrypted_ack(res)
    if not encrypted_ack:
        log.warning("[%s] MHA ack: no encrypted body found in gateway reply",
                    p.acknowledgement_no)
        return

    try:
        ack = encryption.decrypt_payload(encrypted_ack)
    except Exception as e:                       # noqa: BLE001
        log.error("[%s] MHA ack decryption failed: %s", p.acknowledgement_no, e)
        return

    # job_id from the ack message; fall back to this complaint's own job_id.
    message = str(ack.get("message", "") or "")
    m = _JOB_ID_RE.search(message)
    job_id = m.group(0) if m else p.job_id
    if m and job_id != p.job_id:
        log.warning("[%s] MHA ack job_id=%s differs from current job_id=%s "
                    "— storing against the ack's job_id",
                    p.acknowledgement_no, job_id, p.job_id)

    row = s.execute(
        select(ApiResponseLog).where(ApiResponseLog.job_id == job_id)
    ).scalar_one_or_none()
    if not row:
        log.warning("[%s] MHA ack: no api_response_logs row for job_id=%s",
                    p.acknowledgement_no, job_id)
        return

    row.acknowledgement_mha = ack
    row.responded_at = _now()
    s.add(row)
    s.flush()
    log.info("[%s] MHA ack stored job_id=%s status=%s/%s",
             p.acknowledgement_no, job_id,
             ack.get("statuscode"), ack.get("status_message"))


def _send_mha_response(s: Session, p: ComplaintProcessing, payload: dict) -> None:
    """Encrypt the response payload and POST it to the gateway's MHA-response
    endpoint — at most once per (acknowledgement_no, job_id).

    The MHA_RESPONSE step row is the lock: a worker must atomically create it
    (or flip it FAILURE -> PENDING) and commit BEFORE sending. A concurrent or
    retried worker that loses that race never sends. A failed send is recorded
    as FAILURE and raised, so the normal retry (attempt_count -> NEEDS_REVIEW)
    re-sends it. A crash mid-send leaves PENDING: the outcome is unknown, so it
    is never re-sent automatically (at-most-once) and ends in NEEDS_REVIEW."""
    idem = f"MHARESP:{p.acknowledgement_no}:{p.job_id}"
    prior = _step_get(s, p, "MHA_RESPONSE", idem)
    if prior and prior.status == "SUCCESS":
        return

    if not encryption.is_enabled():
        log.warning("[%s] MHA response NOT sent: MHA_RESPONSE_KEY_B64 unset (dev mode)",
                    p.acknowledgement_no)
        return

    encrypted = encryption.encrypt_payload(payload)   # raises -> retried, nothing claimed

    step = _claim_mha_send(s, p, idem)
    if step is None:
        cur = _step_get(s, p, "MHA_RESPONSE", idem)
        if cur:
            s.refresh(cur)                       # another worker may have just finished
        if cur and cur.status == "SUCCESS":
            return
        raise RuntimeError("MHA response is being sent by another worker, or an "
                           "earlier send's outcome is unknown")

    try:
        res = gateway().mha_response(encrypted)
    except GatewayError as e:
        log.error("[%s] MHA response send failed (will retry): %s", p.acknowledgement_no, e)
        step.status, step.response_payload = "FAILURE", {"error": str(e)}
        s.commit()
        raise
    step.status, step.response_payload = "SUCCESS", res
    s.commit()                                   # record delivery before anything else
    log.info("[%s] MHA response sent OK", p.acknowledgement_no)
    _store_mha_ack(s, p, res)


def _claim_mha_send(s: Session, p: ComplaintProcessing,
                    idem: str) -> Optional[ProcessingStep]:
    """Atomically take the right to send; returns the PENDING row we now own, or
    None. The UNIQUE idempotency_key settles a first-send race; the conditional
    UPDATE settles a retry race."""
    step: Optional[ProcessingStep] = ProcessingStep(
        complaint_id=p.complaint_id, acknowledgement_no=p.acknowledgement_no,
        job_id=p.job_id, step="MHA_RESPONSE", idempotency_key=idem, status="PENDING",
        request_payload={"acknowledgement_no": p.acknowledgement_no}, response_payload={},
    )
    try:
        with s.begin_nested():
            s.add(step)
    except IntegrityError:
        retaken = s.execute(
            update(ProcessingStep)
            .where(ProcessingStep.idempotency_key == idem, ProcessingStep.status == "FAILURE")
            .values(status="PENDING", attempt=ProcessingStep.attempt + 1)
        ).rowcount == 1
        step = _step_get(s, p, "MHA_RESPONSE", idem) if retaken else None
    s.commit()
    return step
